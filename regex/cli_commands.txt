CREATING A FOLDER AND COPYING N RANDOM OTHER FOLDERS INTO IT
mkdir <folder-name>                                                                # in ssh terminal
ls | shuf -n N | while read file; do cp "$file" <folder-name>/; done               # in ssh terminal

COUNT THE NUMBER OF ITEMS (FILES AND FOLDERS) IN A DIRECTORY
ls -1 | wc -1       # while in the directory
